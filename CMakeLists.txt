cmake_minimum_required(VERSION 3.22)
project(SillyPlayer VERSION 1.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
if(NOT CMAKE_CROSSCOMPILING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.deps/root/usr")
  list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/.deps/root/usr")
  list(PREPEND CMAKE_INCLUDE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/.deps/root/usr/include")
  list(PREPEND CMAKE_LIBRARY_PATH "${CMAKE_CURRENT_SOURCE_DIR}/.deps/root/usr/lib/x86_64-linux-gnu")
endif()
set(SILLY_QT_COMPONENTS Core Gui Quick QuickControls2 Widgets Sql Concurrent Network OpenGL)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  list(APPEND SILLY_QT_COMPONENTS DBus)
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS ${SILLY_QT_COMPONENTS})
find_package(OpenSSL REQUIRED)
find_path(MPV_INCLUDE_DIR mpv/client.h REQUIRED)
find_library(MPV_LIBRARY NAMES mpv libmpv.so.2 REQUIRED)
find_path(TAG_INCLUDE_DIR taglib/fileref.h REQUIRED)
find_library(TAG_LIBRARY NAMES tag libtag.so.1 REQUIRED)
find_path(ARCHIVE_INCLUDE_DIR archive.h REQUIRED)
find_library(ARCHIVE_LIBRARY NAMES archive libarchive.so.13 REQUIRED)
add_library(sillycore STATIC src/store.cpp src/sync.cpp src/third_party/qrcodegen.cpp src/listening.cpp src/library.cpp src/librarytools.cpp src/player.cpp src/online.cpp src/youtubeplaylist.cpp src/directtransfer.cpp src/extraction.cpp src/catalogue.cpp src/app.cpp src/backup.cpp src/discordrpc.cpp src/sleeptimer.h src/stablemodel.h)
target_compile_definitions(sillycore PRIVATE SILLYPLAYER_VERSION="${PROJECT_VERSION}")
target_include_directories(sillycore PUBLIC src ${MPV_INCLUDE_DIR} ${TAG_INCLUDE_DIR} ${ARCHIVE_INCLUDE_DIR})
target_link_libraries(sillycore PUBLIC Qt6::Core Qt6::Gui Qt6::Sql Qt6::Concurrent Qt6::Network Qt6::Widgets ${MPV_LIBRARY} ${TAG_LIBRARY} ${ARCHIVE_LIBRARY} OpenSSL::Crypto)
target_compile_options(sillycore PRIVATE -Wall -Wextra -Wpedantic)
add_executable(sillyplayer src/main.cpp resources.qrc)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_sources(sillyplayer PRIVATE src/mpris.h)
  target_link_libraries(sillycore PUBLIC Qt6::DBus)
endif()
if(WIN32)
  enable_language(RC)
  target_sources(sillyplayer PRIVATE assets/windows.rc)
  target_compile_definitions(sillycore PUBLIC NOMINMAX UNICODE _UNICODE)
  set_target_properties(sillyplayer PROPERTIES WIN32_EXECUTABLE TRUE AUTORCC_OPTIONS "--compress-algo;zlib")
endif()
target_link_libraries(sillyplayer PRIVATE sillycore Qt6::Quick Qt6::QuickControls2 Qt6::OpenGL)
target_compile_definitions(sillyplayer PRIVATE SILLYPLAYER_VERSION="${PROJECT_VERSION}")
# resolve the bundled runtime even when bin/sillyplayer is invoked directly
# DT_RPATH also applies to indirect dependencies (qt plugins and libmpv codecs)
set_target_properties(sillyplayer PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/../lib")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(X11 REQUIRED)
  target_link_libraries(sillyplayer PRIVATE X11::X11)
  target_compile_definitions(sillyplayer PRIVATE SILLYPLAYER_X11)
  target_link_options(sillyplayer PRIVATE "LINKER:--disable-new-dtags")
endif()
install(TARGETS sillyplayer RUNTIME DESTINATION bin)

include(CTest)
if(BUILD_TESTING)
  add_executable(desktop_tests tests/desktop.cpp resources.qrc)
  target_link_libraries(desktop_tests PRIVATE sillycore Qt6::Quick Qt6::QuickControls2)
  add_test(NAME desktop_regressions COMMAND desktop_tests)
  set_tests_properties(desktop_regressions PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
