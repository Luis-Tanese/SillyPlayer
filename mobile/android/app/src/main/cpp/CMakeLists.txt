cmake_minimum_required(VERSION 3.22)
project(sillymedia LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
set(NATIVE_ROOT "${PROJECT_ROOT}/.deps/android-native/${ANDROID_ABI}")
add_library(sillymedia SHARED native.cpp)
target_include_directories(sillymedia PRIVATE "${PROJECT_ROOT}/src")
target_include_directories(sillymedia PRIVATE "${NATIVE_ROOT}/include" "${PROJECT_ROOT}/.deps/android-native/include")
foreach(component avformat avcodec swresample avutil mp3lame ssl crypto tag)
    add_library(native_${component} STATIC IMPORTED)
    set_target_properties(native_${component} PROPERTIES IMPORTED_LOCATION "${NATIVE_ROOT}/lib/lib${component}.a")
endforeach()
target_link_libraries(sillymedia PRIVATE native_avformat native_avcodec native_swresample native_avutil native_mp3lame native_ssl native_crypto native_tag z dl log android)
target_link_options(sillymedia PRIVATE "-Wl,-Bsymbolic" "-Wl,--exclude-libs,ALL" "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
